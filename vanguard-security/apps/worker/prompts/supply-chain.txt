<systematic_approach>
**Step 1 — Manifest Discovery**
```bash
# Find all package manifests in the repo
find {{REPO_PATH}} -maxdepth 4 \
  \( -name "package.json" -o -name "package-lock.json" -o -name "yarn.lock" \
  -o -name "requirements.txt" -o -name "Pipfile.lock" -o -name "poetry.lock" \
  -o -name "go.mod" -o -name "go.sum" -o -name "Cargo.toml" -o -name "Gemfile.lock" \) \
  ! -path "*/node_modules/*" ! -path "*/.git/*" 2>/dev/null
```

**Step 2 — npm/Node.js Audit**
```bash
cd {{REPO_PATH}}
# Run npm audit in JSON mode (reads package-lock.json — does NOT install)
npm audit --json 2>/dev/null | python3 -c "
import json,sys
d=json.load(sys.stdin)
vulns=d.get('vulnerabilities',{})
for name,v in list(vulns.items())[:20]:
  print(f'{v[\"severity\"].upper()}: {name} — {v.get(\"title\",\"CVE\")} ({v.get(\"url\",\"\")})')
" 2>/dev/null | head -50
```

**Step 3 — Python Dependency Audit**
```bash
# pip-audit (skip if not installed)
if command -v pip-audit >/dev/null 2>&1; then
  pip-audit -r {{REPO_PATH}}/requirements.txt --format json 2>/dev/null | head -100
elif [ -f {{REPO_PATH}}/requirements.txt ]; then
  echo "pip-audit not available — manual CVE check for top packages:"
  head -20 {{REPO_PATH}}/requirements.txt
fi
```

**Step 4 — Go Module Audit**
```bash
if [ -f {{REPO_PATH}}/go.mod ]; then
  cd {{REPO_PATH}} && go list -json -m all 2>/dev/null | \
    python3 -c "import json,sys; [print(d.get('Path',''),d.get('Version','')) for d in [json.loads(l) for l in sys.stdin.read().split('\n}\n') if l.strip()] if isinstance(d,dict)]" 2>/dev/null | head -30
  # govulncheck (skip if not installed)
  command -v govulncheck >/dev/null && govulncheck ./... 2>&1 | head -50
fi
```

**Step 5 — Dependency Confusion Risk Assessment**
Check for internal/scoped package names that could be hijacked on public registries:
```bash
# Find scoped packages and check if they exist on public npm
cd {{REPO_PATH}} && python3 -c "
import json, subprocess, sys
try:
  pkg = json.load(open('package.json'))
  deps = {**pkg.get('dependencies',{}), **pkg.get('devDependencies',{})}
  internal = [n for n in deps if n.startswith('@') and 'internal' in n.lower() or 'private' in n.lower()]
  for p in internal[:5]:
    print(f'Checking {p}...')
    result = subprocess.run(['npm', 'show', p, 'version'], capture_output=True, text=True, timeout=10)
    print(f'  Public registry: {result.stdout.strip() or \"NOT FOUND (confusion risk)\"}')
except Exception as e:
  print(f'Error: {e}')
" 2>/dev/null
```

**Step 6 — Typosquatting Detection**
Search for common typosquatting patterns of top-50 npm packages in the dependency list:
```bash
python3 -c "
# Common typosquatted npm package patterns
known_squats = ['lodahs', 'requst', 'expres', 'mongoos', 'axios-http', 'react-dom-server', 'node-fetch2', 'cross-fetch2']
try:
  import json
  pkg = json.load(open('{{REPO_PATH}}/package.json'))
  deps = list({**pkg.get('dependencies',{}), **pkg.get('devDependencies',{})}.keys())
  hits = [d for d in deps if d in known_squats]
  print('Typosquatting candidates:', hits or 'None found')
  print('Total deps:', len(deps))
except Exception as e:
  print('Error:', e)
" 2>/dev/null
```

**Step 7 — Outdated Dependency Surface**
```bash
# Quick summary of severely outdated packages (major version behind)
cd {{REPO_PATH}} && npm outdated --json 2>/dev/null | python3 -c "
import json,sys
d=json.load(sys.stdin)
major_behind = {k:v for k,v in d.items() if v.get('current','0').split('.')[0] != v.get('latest','0').split('.')[0]}
for k,v in list(major_behind.items())[:15]:
  print(f'  {k}: {v.get(\"current\")} → {v.get(\"latest\")}')
print(f'Total major-version-behind: {len(major_behind)}')
" 2>/dev/null | head -30
```

**Step 8 — Critical CVE Review**
For any Critical or High severity CVEs found in Steps 2-4, look up the CVE details:
```bash
# Check NVD for specific CVEs
for cve in CVE-XXXX-YYYY; do
  curl -s "https://services.nvd.nist.gov/rest/json/cves/2.0?cveId=${cve}" 2>/dev/null | \
    python3 -c "import json,sys; d=json.load(sys.stdin); [print(v['cve']['id'], v['cve']['descriptions'][0]['value'][:200]) for v in d.get('vulnerabilities',[])]" 2>/dev/null
done
```
